find_package(PkgConfig REQUIRED)
pkg_check_modules(COMPACT_JSONC REQUIRED IMPORTED_TARGET json-c)
pkg_check_modules(COMPACT_SQLITE REQUIRED IMPORTED_TARGET sqlite3)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(compact-scan scan.c prepare.c)
set_target_properties(compact-scan PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(compact-scan PRIVATE HAVE_CONFIG_H)
target_compile_options(compact-scan PRIVATE -Wall -Wextra -Werror)
target_include_directories(compact-scan PRIVATE "${CMAKE_BINARY_DIR}" "${CMAKE_SOURCE_DIR}/libclamav" "${CMAKE_SOURCE_DIR}/libfreshclam")
target_link_libraries(compact-scan PRIVATE ClamAV::libclamav PkgConfig::COMPACT_JSONC PkgConfig::COMPACT_SQLITE OpenSSL::Crypto ZLIB::ZLIB ${CMAKE_DL_LIBS})
add_executable(compact-update update.c)
set_target_properties(compact-update PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(compact-update PRIVATE HAVE_CONFIG_H)
target_compile_options(compact-update PRIVATE -Wall -Wextra -Werror)
target_include_directories(compact-update PRIVATE "${CMAKE_BINARY_DIR}" "${CMAKE_SOURCE_DIR}/libclamav" "${CMAKE_SOURCE_DIR}/libfreshclam")
target_link_libraries(compact-update PRIVATE ClamAV::libclamav ClamAV::libfreshclam PkgConfig::COMPACT_JSONC OpenSSL::Crypto)
set_target_properties(compact-scan compact-update PROPERTIES BUILD_RPATH "${CMAKE_BINARY_DIR}/libclamav;${CMAKE_BINARY_DIR}/libfreshclam" INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS compact-scan compact-update RUNTIME DESTINATION bin)
